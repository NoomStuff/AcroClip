cmake_minimum_required(VERSION 3.24)
project(acroclip_native_fixture LANGUAGES CXX)
if(WIN32)
add_executable(acroclip-recording-size-test recording-size.cpp)
target_compile_features(acroclip-recording-size-test PRIVATE cxx_std_17)
target_compile_options(acroclip-recording-size-test PRIVATE /UNDEBUG)
add_executable(acroclip-capture-window-test capture-window.cpp)
target_compile_features(acroclip-capture-window-test PRIVATE cxx_std_17)
target_compile_options(acroclip-capture-window-test PRIVATE /UNDEBUG)
target_link_libraries(acroclip-capture-window-test PRIVATE user32 dwmapi)
add_executable(acroclip-d3d-fixture d3d-window.cpp)
target_compile_features(acroclip-d3d-fixture PRIVATE cxx_std_17)
target_link_libraries(acroclip-d3d-fixture PRIVATE d3d11 dxgi user32 winmm)
target_compile_definitions(acroclip-d3d-fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(UNIX AND NOT APPLE)
  find_package(X11 REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PULSE_SIMPLE REQUIRED IMPORTED_TARGET libpulse-simple)
  pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
  add_executable(acroclip-pulse-client-control pulse-client-control.cpp)
  target_compile_features(acroclip-pulse-client-control PRIVATE cxx_std_17)
  target_link_libraries(acroclip-pulse-client-control PRIVATE PkgConfig::PULSE)
  add_executable(acroclip-pulse-sync-fixture pulse-sync.cpp)
  target_compile_features(acroclip-pulse-sync-fixture PRIVATE cxx_std_17)
  target_link_libraries(acroclip-pulse-sync-fixture PRIVATE X11::X11 PkgConfig::PULSE_SIMPLE)
  add_executable(acroclip-x11-audio-fixture x11-audio.cpp)
  target_compile_features(acroclip-x11-audio-fixture PRIVATE cxx_std_17)
  target_link_libraries(acroclip-x11-audio-fixture PRIVATE X11::X11 PkgConfig::PULSE_SIMPLE)
  find_program(MINGW_C_COMPILER x86_64-w64-mingw32-gcc)
  if(MINGW_C_COMPILER)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/AcroClipWineGame.exe"
      COMMAND "${MINGW_C_COMPILER}" -O2 -static "${CMAKE_CURRENT_SOURCE_DIR}/wine-window.c"
        -o "${CMAKE_CURRENT_BINARY_DIR}/AcroClipWineGame.exe" -luser32 -lgdi32 -lwinmm
      DEPENDS wine-window.c VERBATIM)
    add_custom_target(acroclip-wine-fixture ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/AcroClipWineGame.exe")
  endif()
else()
  message(FATAL_ERROR "No native fixture for this platform")
endif()
